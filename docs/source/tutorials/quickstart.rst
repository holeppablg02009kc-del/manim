# -*- coding: utf-8 -*-
"""
ನಾಟಕದ ಹೆಸರು: ಜೀವನದ ದಾರಿ (ರವಿ ಬೆಳಗೇರೆ ಅವರ ಶೈಲಿಯ ಪ್ರೇರಣೆಯಿಂದ)
ಪಾತ್ರಗಳು: 
1. ರವಿ (ಪತ್ರಕರ್ತ / ನಿರೂಪಕ)
2. ಚೈತನ್ಯ (ಯುವಕ, ಜೀವನದಲ್ಲಿ ಜಿಗುಪುಮೆಗೊಂಡವನು)
"""

class KannadaDrama:
    def __init__(self, title, author_inspiration):
        self.title = title
        self.author_inspiration = author_inspiration

    def display_intro(self):
        print(f"=== ನಾಟಕ: {self.title} ===")
        print(f"ಪ್ರೇರಣೆ: {self.author_inspiration}\n")

    def play_scene_1(self):
        print("--- ದೃಶ್ಯ 1: ಸಂಜೆಯ ವೇಳೆ ಒಂದು ಹಳೆಯ ಕಾಫಿ ಹೌಸ್ ---")
        print("ಸ್ಥಳ: ಮೇಜಿನ ಮೇಲೆ ಅರ್ಧ ಕುಡಿದ ಕಾಫಿ ಕಪ್ ಮತ್ತು ಹರಡಿದ ಕಾಗದಗಳು.\n")
        
        dialogues = [
            ("ರವಿ", "ಬೆಳಕು ನಮಗೆ ಕಾಣಿಸೋದು ಕತ್ತಲು ಕರಗಿದ ಮೇಲಲ್ಲಪ್ಪಾ... ಕತ್ತಲಲ್ಲೇ ಒಂದು ದೀಪ ಹಚ್ಚುವ ಧೈರ್ಯ ಬಂದಾಗ! ಯಾಕೆ ಹೀಗೆ ಕುಗ್ಗಿದ್ದೀಯಾ?"),
            ("ಚೈತನ್ಯ", "ಸರ್, ಬದುಕು ಅಂದುಕೊಂಡಂತೆ ನಡೆಯುತ್ತಿಲ್ಲ. ಎಲ್ಲವೂ ಕೈ ತಪ್ಪಿ ಹೋಗುತ್ತಿದೆ ಎಂಬ ಭಾವನೆ ಕಾಡುತ್ತಿದೆ. ಸೋಲು ನನ್ನನ್ನು ಆವರಿಸಿದೆ."),
            ("ರವಿ", "ಸೋಲು ಅನ್ನೋದು ಒಂದು ನಿಲ್ದಾಣವಲ್ಲ, ಅದೊಂದು ದಾರಿ ಅಷ್ಟೇ. ಬದುಕಿನ ಒರತೆ ಇರೋದೇ ಹೋರಾಟದಲ್ಲಿ. ನಿನ್ನೊಳಗಿನ ನಂಬಿಕೆಯನ್ನು ಎಂದಿಗೂ ಸಾಯಲು ಬಿಡಬೇಡ.")
        ]
        
        for character, dialogue in dialogues:
            print(f"{character}: \"{dialogue}\"")
        print("\n[ಹಿನ್ನೆಲೆಯಲ್ಲಿ ಮೃದುವಾದ ಮಳೆಯ ಶಬ್ದ ಮತ್ತು ಹಳೆಯ ರೇಡಿಯೋ ಹಾಡಿನ ಗುನುಗು ಕೇಳಿಸುತ್ತದೆ]\n")

    def play_scene_2(self):
        print("--- ದೃಶ್ಯ 2: ಅದೇ ಸ್ಥಳ, ಸ್ವಲ್ಪ ಸಮಯದ ನಂತರ ---")
        
        dialogues = [
            ("ಚೈತನ್ಯ", "(ನಿಟ್ಟುಸಿರುಬಿಡುತ್ತಾ) ನಿಮ್ಮ ಮಾತುಗಳನ್ನು ಕೇಳಿದ ಮೇಲೆ ಮನಸ್ಸಿಗೆ ಸ್ವಲ್ಪ ಸಮಾಧಾನವಾಯಿತು ಸರ್. ಕತ್ತಲೆಯನ್ನು ಹಳಿಯುವುದಕ್ಕಿಂತ ಒಂದು ಚಿಕ್ಕ موم بتی (ದೀಪ) ಹಚ್ಚುವುದು ಮೇಲು ಅನ್ನಿಸುತ್ತಿದೆ."),
            ("ರವಿ", "ಖಂಡಿತ! ಮನುಷ್ಯನ ಮನಸ್ಸು ಒಂದು ಅದ್ಭುತ ಶಕ್ತಿ. ಬರೆಯುವ ಲೇಖನವಾಗಲಿ ಅಥವಾ ಬದುಕಿನ ಹಾದಿಯಾಗಲಿ, ನಮ್ಮದೇ ಆದ ಸಹಿಯಿರಬೇಕು."),
            ("ಚೈತನ್ಯ", "ಸರಿ ಸರ್, ಇನ್ನು ಹಿಂದೆ ತಿರುಗಿ ನೋಡುವುದಿಲ್ಲ. ಹೊಸ ಹೋರಾಟವನ್ನು ಮತ್ತೆ ಆರಂಭಿಸುತ್ತೇನೆ.")
        ]
        
        for character, dialogue in dialogues:
            print(f"{character}: \"{dialogue}\"")
        print("\n=== ನಾಟಕ ಮುಕ್ತಾಯ ===")

# ನಾಟಕವನ್ನು ರನ್ ಮಾಡುವ ಕೋಡ್
if __name__ == "__main__":
    drama = KannadaDrama(
        title="ಬೆಳಕಿನ ಹುಡುಕಾಟ", 
        author_inspiration="ರವಿ ಬೆಳಗೇರೆ ಅವರ ಸ್ಫೂರ್ತಿದಾಯಕ ಬರಹಗಳು"
    )
    
    drama.display_intro()
    drama.play_scene_1()
    drama.play_scene_2()
